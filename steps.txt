```javascript
window.playVideo = function () {
  const modal = document.getElementById('videoModal');
  const videoContainer = document.getElementById('modal-video-container');
  const placeholder = document.getElementById('video-placeholder');
  const closeBtn = document.getElementById('closeBtn');

  // Hide placeholder
  placeholder.style.display = 'none';

  // Create video element for local video
  videoContainer.innerHTML = `
    <video id="course-video" class="course-video" controls autoplay>
      <source src="{{ course.promo_video.url }}" type="video/mp4">
      {% if course.promo_video_webm %}
      <source src="{{ course.promo_video_webm.url }}" type="video/webm">
      {% endif %}
      Your browser does not support the video tag.
    </video>
  `;

  // Show modal
  modal.style.display = 'flex';

  // Track video play event
  trackVideoPlay('{{ course.id }}', 'preview');

  // Close modal
  closeBtn.addEventListener('click', () => {
    modal.style.display = 'none';
    videoContainer.innerHTML = '';
  });

  // Close modal when clicking outside
  modal.addEventListener('click', (e) => {
    if (e.target === modal) {
      modal.style.display = 'none';
      videoContainer.innerHTML = '';
    }
  });

  // Video ended event tracking
  $(document).on('ended', '#course-video', function () {
    trackVideoPlay('{{ course.id }}', 'preview_completed');
  });
};

window.playExternalVideo = function (videoUrl) {
  const modal = document.getElementById('videoModal');
  const videoContainer = document.getElementById('modal-video-container');
  const placeholder = document.getElementById('video-placeholder');
  const closeBtn = document.getElementById('closeBtn');

  // Hide placeholder
  placeholder.style.display = 'none';

  // Handle different video platforms
  if (videoUrl.includes('youtube.com') || videoUrl.includes('youtu.be')) {
    const videoId = extractYouTubeId(videoUrl);
    videoContainer.innerHTML = `
      <iframe 
        width="100%" 
        height="100%" 
        src="https://www.youtube.com/embed/${videoId}?autoplay=1" 
        frameborder="0" 
        allowfullscreen>
      </iframe>
    `;
  } else if (videoUrl.includes('vimeo.com')) {
    const videoId = extractVimeoId(videoUrl);
    videoContainer.innerHTML = `
      <iframe 
        width="100%" 
        height="100%" 
        src="https://player.vimeo.com/video/${videoId}?autoplay=1" 
        frameborder="0" 
        allowfullscreen>
      </iframe>
    `;
  } else {
    videoContainer.innerHTML = `
      <video width="100%" height="100%" controls autoplay>
        <source src="${videoUrl}" type="video/mp4">
        Your browser does not support the video tag.
      </video>
    `;
  }

  // Show modal
  modal.style.display = 'flex';

  // Track external video play
  trackVideoPlay('{{ course.id }}', 'external_preview');

  // Close modal
  closeBtn.addEventListener('click', () => {
    modal.style.display = 'none';
    videoContainer.innerHTML = '';
  });

  // Close modal when clicking outside
  modal.addEventListener('click', (e) => {
    if (e.target === modal) {
      modal.style.display = 'none';
      videoContainer.innerHTML = '';
    }
  });
};
```
